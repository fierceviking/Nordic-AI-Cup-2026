# Apple Silicon (M-series) dependencies. Use this instead of requirements.txt,
# which installs torch from the CUDA index and faster-whisper.
#
# CTranslate2, which faster-whisper runs on, has no Metal backend -- its GPU
# support is cuBLAS and ROCm only -- so large-v3 falls back to CPU and needs an
# estimated 66-145 s for a two minute conversation, against a 60 s budget. ASR
# runs through MLX here instead.
#
#   pyenv virtualenv 3.11.13 dm
#   pyenv activate dm
#   brew install ffmpeg          # mlx-whisper and parakeet-mlx shell out to it
#   pip install -r requirements-mac.txt
#
# Verified on Python 3.11.13, macOS 26.6, M3 Pro.
